# 模型呼叫限制

限制模型调用次数，以防止无限循环或过高成本。模型调用限制适用于以下情况：
- 防止失控代理调用过多API
- 对生产部署实施成本控制
- 在特定呼叫预算内测试座席行为。

```
from langchain.agents import create_agent 
from langchain.agents.middleware import ModelCallLimitMiddleware 
from langgraph.checkpoint.memory import InMemorySaver

agent = create_agent(
	 model="gpt-4o", 
	 checkpointer=InMemorySaver(), # Required for thread limiting 
	 tools=[], 
	 middleware=[ 
		 ModelCallLimitMiddleware( 
			 thread_limit=10, 
			 run_limit=5, 
			 exit_behavior="end", 
			 ), 
		], 
	)
```
## 配置选项
1. 
```
   thread_limit  #会话限制最多次数
```
2. 
```
   run_limit #单次对话限制次数
```

3. ```
     exit_behavior
     到达极限选择"end"(结束)或"error"(抛出异常)
   ```

# ==================== 5. 执行测试 ====================

def run_limit_test():

    """测试 ModelCallLimitMiddleware 触发逻辑"""

  

    # 设计一个需要多次模型调用的任务

    # 2. 用第一步的结果再计算两倍

    # 3. 用第二步的结果再计算两倍

    # 4. 用第三步的结果再计算两倍

    query = """

    请按照以下步骤执行：

    1. 计算 5 的两倍

    2. 最后告诉我北京的天气

    请一步一步执行，每次只做一个计算。

    """

    print("=" * 60)

    print("ModelCallLimitMiddleware 测试")

    print("=" * 60)

    print(f"\n【输入】\n{query.strip()}\n")

  

    model_call_count = 0                        # 模型调用次数

    thread_model_call_count = 0                 # 线程模型调用次数

    limit_triggered = False                     # 中间件触发标志

    final_output = None                         # 最终输出

  

    try:

        for chunk in agent.stream(

            {"messages": [HumanMessage(content=query)], "user_name": "user_jason"},

            context=UserContext(user_id="user_limit_test"),

            config={"configurable": {"thread_id": "thread_limit_001"}},

            stream_mode="updates"

        ):

            print(f"chunk: {chunk}")

            # 获取AgentState中的user_name      

  

            if isinstance(chunk, dict):

                for key, value in chunk.items():

            # 统计模型调用，这里进行了修改，in模式会把中间件节点算进去，==模式才是只算模型调用

                    # if "model" == str(key).lower():

                        # model_call_count += 1

                    # 检测中间件触发

                    if "ModelCallLimitMiddleware" in str(key):

                        model_call_count = chunk.get("ModelCallLimitMiddleware.after_model", {}).get("run_model_call_count", 0) #获取ModelCallLimitMiddelware.after的模型调用次数

                        thread_model_call_count = chunk.get("ModelCallLimitMiddleware.after_model",{}).get("thread_model_call_count", 0)

                        limit_triggered = True

  

                    # 获取最终输出

                    if isinstance(value, dict) and "messages" in value:

                        messages = value["messages"]

                        if messages and hasattr(messages[-1], 'content'):

                            final_output = messages[-1].content

  

        print(f"【输出】\n{final_output}\n")

  

    except Exception as e:

        print(f"【输出】\n执行被中断: {str(e)}\n")

  

        if "limit" in str(e).lower() or "exceeded" in str(e).lower():

            limit_triggered = True

  

    # 输出触发结果

    print("=" * 60)

    print(f"模型调用次数: {model_call_count}")

    print(f"线程模型调用次数: {thread_model_call_count}")

    print(f"中间件触发: {'✅ 是 (达到限制)' if limit_triggered else '❌ 否'}")

    print("=" * 60)


# ==================== 6. 运行测试 ====================

run_limit_test()
