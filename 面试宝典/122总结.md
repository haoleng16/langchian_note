```
def get_retriever():
    """
    构建检索器
    :param vector_store: 向量数据库
    :return: 检索器
    """
    #1. 获取向量数据库
    vector_store = FAISS.load_local(
        "../faiss/faiss_simple_db",                         # 向量数据库文件路径
        embedding,                                          # 嵌入模型
        allow_dangerous_deserialization = True              # 允许反序列化
    )

    #2. 创建检索器
    retriever = vector_store.as_retriever(
        search_kwargs={"k": 3},                             # 检索Top3相关分片
        search_type = "similarity"                          # 使用相似度检索
    )
    print(f"✅ 检索器构建成功!")
    return retriever
```
## 3.用户问题向量化
```
# 这就是 retriever.invoke() 内部隐藏的逻辑流
class VectorStoreRetriever:
    def invoke(self, query_text):
        # 1. 【自动向量化】
        # 这里的 self.embedding 就是你在 load_local 时传入的那个模型
        # 它自动调用 embed_query 方法，把中文问题变成 [0.1, 0.5, ...]
        query_vector = self.vector_store.embedding.embed_query(query_text)

        # 2. 【向量检索】
        # 拿着生成的向量去 FAISS 索引里找
        docs = self.vector_store.similarity_search_by_vector(query_vector, k=3)

        return docs
```

# 构建索引的过程落实到代码
## 1.利用数据加载器加载文档并返回文档列表
```
def load_documents(file_paths: str) -> list[Document]:
    """
    加载文档
    :param file_paths: 文件路径列表
    :return: 文档列表
    """
    #1. 加载txt文档
    base_loader = TextLoader(file_paths, encoding="utf-8")
    documents = base_loader.load()
    return documents
```
## 2.创建文本切割器(并设置块的大小和重叠大小和分割符)
```
def split_documents(documents: str) -> list[Document]:
    """
    文档切片
    :param documents: 文档列表
    :return: 文档列表
    """
    #1. 创建文本切割器
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size = 150,                                   # 每个块的大小
        chunk_overlap = 10,                                 # 块之间的重叠大小
        separators = ["\n\n", "\n", " ", ""]                # 按照换行符,空格等符号进行切割
    )
```
## 3.文档分块
```
#2. 切割文档
    chunks = text_splitter.split_documents(documents)
    return chunks
```

# 大模型存在的缺陷，怎么解决？使用RAG的优势在哪里？
1. 缺陷1:大模型的幻觉问题 :  
2. 知识滞后性,缺乏时效性
3. 可解释性能力弱,没有来源追溯
4. 上下文窗口太长容易忽视关键词


怎么解决?
RAG 通过强制模型从知识库中进行检索,基于检索到的真实信息生成内容

但是还有模型微调

模型微调跟RAG比,使用RAG的优势在哪里?
1. RAG不用动模型,只要更新知识库几分钟生效,微调速度根本跟不上
2. 模型微调成本高,需要海量的标注数据,而且微调时间通常要几天甚至几周
3. 并且模型微调还有缺陷容易学杂,出现捡了芝麻丢了西瓜的问题

# 6.RAG能够完全解决大模型的幻觉问题吗?
肯定不能完全解决,RAG只能说大幅降低大模型的幻觉问题
为什么?
1. **文档数据本身是错的**:RAG本身数据清洗时噪声信息存入幻觉问题,那么通过检索出来的信息也不准确
2. **检索召回不足大模型瞎编填补空白**:文档分块检索不全,检索的内容不完全,
3. **召回错误**:召回不相关内容导致模型回答不准确
